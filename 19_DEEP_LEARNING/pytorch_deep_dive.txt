A tensor is a multi-dimensional array (like NumPy array, but GPU-capable).

    import torch

    # Creation
    x = torch.tensor([1, 2, 3])                # From list
    x = torch.zeros(3, 4)                       # Zeros
    x = torch.ones(3, 4)                        # Ones
    x = torch.randn(3, 4)                       # Random normal
    x = torch.arange(0, 10)                     # Range
    x = torch.linspace(0, 1, 5)                 # Evenly spaced

    # Properties
    x.shape          # torch.Size([3, 4])
    x.dtype          # torch.float32
    x.ndim           # 2
    x.device         # cpu or cuda:0

    # NumPy conversion
    arr = x.numpy()              # Tensor → NumPy
    t = torch.from_numpy(arr)    # NumPy → Tensor

    # GPU
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    x_gpu = x.to(device)
    x_cpu = x_gpu.cpu()

OPERATIONS:
    # Same as NumPy
    x + y, x * y, x @ y         # Element-wise and matmul
    x.mean(), x.sum(), x.std()  # Aggregation
    x.reshape(3, -1)             # Reshape
    x.T                          # Transpose

2. AUTOGRAD — AUTOMATIC DIFFERENTIATION
================================================================================

PyTorch tracks operations and can compute gradients automatically.

    x = torch.tensor(2.0, requires_grad=True)
    y = x ** 2 + 3 * x + 1    # y = x² + 3x + 1
    y.backward()               # Compute gradients
    print(x.grad)              # dy/dx = 2x + 3 = 7.0

HOW IT WORKS:
    torch builds a COMPUTATION GRAPH in the forward pass.
    During .backward(), it traverses the graph backwards
    computing gradients using the chain rule.

    This is EXACTLY backpropagation in neural networks.

FOR GRADIENT DESCENT:
    x = torch.tensor(5.0, requires_grad=True)
    lr = 0.01

    for step in range(100):
        y = (x - 3) ** 2          # Loss: we want x → 3
        y.backward()
        x.data -= lr * x.grad     # Update x
        x.grad.zero_()            # Reset gradients

    print(f"x = {x.item():.4f}")  # Should be close to 3

class MLP(nn.Module):
    """Multi-layer perceptron for classification."""

    def __init__(self, input_dim, hidden_dim, num_classes):
        super().__init__()
        self.network = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(hidden_dim, num_classes)
        )

    def forward(self, x):
        return self.network(x)

COMMON LAYERS:

5. COMPLETE TRAINING LOOP
================================================================================

def train_model(model, train_loader, val_loader, epochs=10, lr=0.001):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = model.to(device)

    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    history = {"train_loss": [], "val_loss": [], "val_acc": []}

    for epoch in range(epochs):
        # Training
        model.train()
        train_loss = 0
        for batch_X, batch_y in train_loader:
            batch_X, batch_y = batch_X.to(device), batch_y.to(device)

            optimizer.zero_grad()           # Reset gradients
            outputs = model(batch_X)        # Forward pass
            loss = criterion(outputs, batch_y)  # Compute loss
            loss.backward()                 # Backward pass
            optimizer.step()                # Update weights

            train_loss += loss.item()

        train_loss /= len(train_loader)

        # Validation
        model.eval()
        val_loss = 0
        correct = 0
        total = 0
        with torch.no_grad():
            for batch_X, batch_y in val_loader:
                batch_X, batch_y = batch_X.to(device), batch_y.to(device)
                outputs = model(batch_X)
                loss = criterion(outputs, batch_y)
                val_loss += loss.item()
                _, predicted = outputs.max(1)
                total += batch_y.size(0)
                correct += predicted.eq(batch_y).sum().item()

        val_loss /= len(val_loader)
        val_acc = correct / total

        history["train_loss"].append(train_loss)
        history["val_loss"].append(val_loss)
        history["val_acc"].append(val_acc)

        print(f"Epoch {epoch+1}/{epochs}: "
              f"train_loss={train_loss:.4f}, val_loss={val_loss:.4f}, "
              f"val_acc={val_acc:.4f}")

    return history

MIXED PRECISION (faster training):
    from torch.cuda.amp import autocast, GradScaler

    scaler = GradScaler()
    for batch_X, batch_y in train_loader:
        batch_X, batch_y = batch_X.to(device), batch_y.to(device)
        optimizer.zero_grad()
        with autocast():
            outputs = model(batch_X)
            loss = criterion(outputs, batch_y)
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
