from collections import deque
def bfs(grid, start):
    queue = deque([start])
    visited = {start}
    while queue:
        i, j = queue.popleft()
        for di, dj in [(1,0),(-1,0),(0,1),(0,-1)]:
            ni, nj = i+di, j+dj
            if valid(ni, nj) and (ni, nj) not in visited:
                visited.add((ni, nj))
                queue.append((ni, nj))
