1. READING CSV
================================================================================

    df = pd.read_csv("data.csv")

COMMON PARAMETERS:
    df = pd.read_csv("data.csv",
                      usecols=["name", "age"],         # Only these columns
                      dtype={"age": int},                # Force column types
                      nrows=1000,                        # First 1000 rows
                      skiprows=[1, 3],                   # Skip specific rows
                      na_values=["N/A", "missing", ""],  # Values to treat as NaN
                      parse_dates=["date_column"],        # Parse dates
                      index_col="id",                     # Set index column
                      sep=";",                            # Custom separator
                      encoding="utf-8")                   # File encoding

WRITING CSV:
    df.to_csv("output.csv", index=False)
    df.to_csv("output.csv", index=False, columns=["name", "age"])

================================================================================
2. READING EXCEL
================================================================================

    df = pd.read_excel("data.xlsx")
    df = pd.read_excel("data.xlsx", sheet_name="Sheet2")
    df = pd.read_excel("data.xlsx", usecols=["A", "C"])

    # Multiple sheets
    sheets = pd.read_excel("data.xlsx", sheet_name=None)  # Dict of DataFrames

    df.to_excel("output.xlsx", index=False)

    NOTE: Requires openpyxl package: pip install openpyxl

================================================================================
3. READING JSON
================================================================================

    # Standard JSON (array of objects)
    df = pd.read_json("data.json")

    # JSON Lines (one JSON object per line)
    df = pd.read_json("data.jsonl", lines=True)

    # Nested JSON → DataFrame
    import json
    with open("data.json") as f:
        data = json.load(f)
    df = pd.json_normalize(data)

    WRITING JSON:
    df.to_json("output.json", orient="records", indent=2)

================================================================================
4. PRACTICAL TIPS
================================================================================

ALWAYS INSPECT AFTER LOADING:
    df = pd.read_csv("data.csv")
    print(f"Shape: {df.shape}")
    print(df.head())
    print(df.dtypes)
    print(df.isnull().sum())

HANDLE ENCODING ISSUES:
    df = pd.read_csv("data.csv", encoding="latin-1")    # European characters
    df = pd.read_csv("data.csv", encoding="cp1252")      # Windows encoding

LARGE FILES:
    # Read in chunks
    chunks = pd.read_csv("large.csv", chunksize=10000)
    for chunk in chunks:
        process(chunk)

    # Read only needed columns
    df = pd.read_csv("large.csv", usecols=["col1", "col2"])
