4. RE — Regular Expressions for Text Cleaning
================================================================================

import re

BASIC PATTERNS:
    r"\d"       # Digit
    r"\w"       # Word character (letter, digit, underscore)
    r"\s"       # Whitespace
    r"[aeiou]"  # Any vowel
    r"[^a-z]"   # Not lowercase letter
    r"+"        # One or more
    r"*"        # Zero or more
    r"?"        # Zero or one
    r"{n}"      # Exactly n
    r"{n,m}"    # Between n and m
    r"^"        # Start of string
    r"$"        # End of string

COMMON OPERATIONS:
    re.findall(r"\d+", "price is 100 dollars")    # ['100']
    re.sub(r"\s+", " ", "too   many   spaces")     # 'too many spaces'
    re.search(r"\d+", "order 42 shipped")          # Match object
    re.split(r"[,;]", "a,b;c,d")                   # ['a', 'b', 'c', 'd']

ML USE — TEXT PREPROCESSING:
    text = "Check out https://example.com! Price: $19.99 #sale"

    # Remove URLs
    clean = re.sub(r"http\S+", "", text)

    # Remove special characters
    clean = re.sub(r"[^a-zA-Z0-9\s]", "", clean)

    # Extract numbers
    numbers = re.findall(r"\d+\.?\d*", text)    # ['19.99']

    # Extract hashtags
    hashtags = re.findall(r"#\w+", text)        # ['#sale']

    # Clean email addresses
    emails = re.findall(r"\b[\w.-]+@[\w.-]+\.\w+\b", text)

5. COLLECTIONS — Specialized Data Structures
================================================================================

from collections import Counter, defaultdict, namedtuple, deque

COUNTER — Count occurrences:
    words = ["cat", "dog", "cat", "bird", "cat", "dog"]
    word_counts = Counter(words)
    print(word_counts.most_common(2))    # [('cat', 3), ('dog', 2)]

    ML USE: Word frequency in NLP, class distribution in classification

DEFAULTDICT — Auto-initialize missing keys:
    groups = defaultdict(list)
    for name, dept in [("Alice", "HR"), ("Bob", "Eng"), ("Charlie", "HR")]:
        groups[dept].append(name)
    # {"HR": ["Alice", "Charlie"], "Eng": ["Bob"]}

    ML USE: Group data by category without checking if key exists

NAMEDTUPLE — Lightweight class:
    Point = namedtuple("Point", ["x", "y"])
    p = Point(3, 4)
    print(p.x, p.y)

    ML USE: Clean way to return multiple named values

DEQUE — Double-ended queue (fast appends/pops from both ends):
    recent = deque(maxlen=10)    # Keep last 10 items
    for value in data:
        recent.append(value)     # Auto-discards old items

    ML USE: Sliding window, recent history, caching

8. PICKLE AND JOBLIB — Model Serialization
================================================================================

import pickle
import joblib

PICKLE:
    # Save
    with open("model.pkl", "wb") as f:
        pickle.dump(model, f)

    # Load
    with open("model.pkl", "rb") as f:
        model = pickle.load(f)

JOBLIB (PREFERRED for scikit-learn):
    # Save
    joblib.dump(model, "model.joblib")

    # Load
    model = joblib.load("model.joblib")

WHY JOBLIB OVER PICKLE:
    - More efficient for large NumPy arrays
    - Better compression
    - Standard for scikit-learn models
    - Same API: dump() and load()

SECURITY WARNING:
    NEVER load pickle/joblib files from untrusted sources.
    They can execute arbitrary code when loaded.

ML USE:
    # Save trained model
    import joblib
    joblib.dump(model, "models/churn_classifier.joblib")

9. ITERTOOLS — Efficient Iteration

import itertools

# Combinations and permutations
list(itertools.combinations([1, 2, 3], 2))    # [(1,2), (1,3), (2,3)]
list(itertools.permutations([1, 2, 3], 2))    # [(1,2), (1,3), ...]

# Chain multiple iterables
list(itertools.chain([1, 2], [3, 4], [5, 6]))    # [1, 2, 3, 4, 5, 6]

# Product (cartesian product)
list(itertools.product([0.01, 0.1, 1.0], [3, 5, 7]))
# [(0.01,3), (0.01,5), ..., (1.0,7)]

ML USE:
    # Hyperparameter grid search
    param_grid = list(itertools.product(
        [0.01, 0.1, 1.0],      # learning rates

import functools

LRU_CACHE — Memoization:
    @functools.lru_cache(maxsize=128)
    def expensive_calculation(n):
        return sum(i**2 for i in range(n))

    # Second call with same n is instant (cached)

PARTIAL — Fix some arguments:
    from functools import partial

    def train_model(model, X, y, lr=0.01):
        model.fit(X, y)
