    def calculate_mse(y_true, y_pred):
        n = len(y_true)
        return sum((t - p)**2 for t, p in zip(y_true, y_pred)) / n

EXERCISE 6.1:
    def safe_divide(a, b):
        try:
            return a / b
        except ZeroDivisionError:
            print("Cannot divide by zero")
            return None
        except TypeError:
            print("Invalid types for division")
            return None
