================================================================================
1. TRIE FUNDAMENTALS
================================================================================

A trie (prefix tree) is a tree where each node represents a character.

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True

    def search(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end

    def starts_with(self, prefix):
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True

Time: O(m) for insert/search/startsWith where m = word length
Space: O(n × m × charset) where n = number of words

IMPLEMENT MAGIC DICTIONARY (LC #676)
--------------------------------------
class MagicDictionary:
    def __init__(self):
        self.trie = Trie()

    def buildDict(self, dictionary):
        for word in dictionary:
            self.trie.insert(word)

    def search(self, searchWord):
        def dfs(node, word, i, changed):
            if i == len(word):
                return node.is_end and changed
            if word[i] in node.children:
                if dfs(node.children[word[i]], word, i + 1, changed):
                    return True
            if not changed:
                for char in node.children:
                    if char != word[i]:
                        if dfs(node.children[char], word, i + 1, True):
                            return True
            return False

        return dfs(self.trie.root, searchWord, 0, False)
