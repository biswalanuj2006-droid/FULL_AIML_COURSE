1. QUEUE FUNDAMENTALS
================================================================================

FIFO (First In, First Out)

Python implementation using collections.deque:
from collections import deque

queue = deque()
queue.append(1)       # enqueue
queue.popleft()       # dequeue
queue[0]              # front
len(queue) == 0       # isEmpty

Time: O(1) for all operations

================================================================================
2. DEQUE (Double-Ended Queue)
================================================================================

Can add/remove from both ends.

from collections import deque

dq = deque()
dq.append(1)          # add to right
dq.appendleft(2)      # add to left
dq.pop()              # remove from right
dq.popleft()          # remove from left

USE FOR: sliding window maximum, monotonic queue

================================================================================
3. MONOTONIC QUEUE
================================================================================

INTUITION
---------
A deque where elements are always in increasing (or decreasing) order.
Used for "sliding window maximum/minimum" problems in O(n).

SLIDING WINDOW MAXIMUM (LC #239)
---------------------------------
from collections import deque

def max_sliding_window(nums, k):
    dq = deque()  # stores indices
    result = []

    for i in range(len(nums)):
        # Remove elements outside window
        while dq and dq[0] < i - k + 1:
            dq.popleft()

        # Remove smaller elements (they'll never be max)
        while dq and nums[dq[-1]] < nums[i]:
            dq.pop()

        dq.append(i)

        if i >= k - 1:
            result.append(nums[dq[0]])

    return result

DRY RUN: nums=[1,3,-1,-3,5,3,6,7], k=3
  i=0: dq=[0]
  i=1: 3>1 → dq=[1]
  i=2: -1<3 → dq=[1,2]; result=[3]
  i=3: -3<-1 → dq=[1,2,3]; result=[3,3]
  i=4: 5>-3,-1,3 → dq=[4]; 0 outside → result=[3,3,5]
  i=5: 3<5 → dq=[4,5]; 1 outside → result=[3,3,5,5]
  i=6: 6>3,5 → dq=[6]; 2 outside → result=[3,3,5,5,6]
  i=7: 7>6 → dq=[7]; 3 outside → result=[3,3,5,5,6,7] ✓

================================================================================
4. CIRCULAR QUEUE
================================================================================

class CircularQueue:
    def __init__(self, k):
        self.queue = [None] * k
        self.head = 0
        self.tail = 0
        self.size = 0
        self.capacity = k

    def enqueue(self, val):
        if self.size == self.capacity:
            return False
        self.queue[self.tail] = val
        self.tail = (self.tail + 1) % self.capacity
        self.size += 1
        return True

    def dequeue(self):
        if self.size == 0:
            return False
        self.head = (self.head + 1) % self.capacity
        self.size -= 1
        return True

    def front(self):
        return self.queue[self.head] if self.size > 0 else -1

    def rear(self):
        return self.queue[(self.tail - 1) % self.capacity] if self.size > 0 else -1

5. BFS WITH QUEUES
================================================================================

from collections import deque

def bfs(graph, start):
    visited = set([start])
    queue = deque([start])
    order = []

    while queue:
        node = queue.popleft()
        order.append(node)
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    return order
